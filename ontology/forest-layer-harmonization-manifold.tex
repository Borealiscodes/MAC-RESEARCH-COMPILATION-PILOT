\section{Manifold Geometry Axioms}

\begin{axiom}[Patch Semantics]
Each altitude patch $A_i \in P$ represents a coordinate chart:
\[
A0 = \text{Directional},\;
A1 = \text{Seam},\;
A2 = \text{Meaning},\;
A3 = \text{Runtime},\;
A4 = \text{Diagnostics},\;
A5 = \text{Falsification}.
\]
\end{axiom}

\begin{axiom}[Transition Map Semantics]
Each membrane transition map $m \in M$ defines a contamination-safe chart map:
\[
\begin{aligned}
Neutral &: A1 \leftrightarrow A2 \\
Formal &: A0 \leftrightarrow A2 \\
Structural &: A2 \leftrightarrow A3 \\
Diagnostic &: A3 \leftrightarrow A4 \\
Provenance &: A4 \leftrightarrow A5
\end{aligned}
\]
\end{axiom}

\begin{axiom}[Atlas Reversibility]
All atlas transitions are reversible:
\[
T_{alt}(A_i, A_j) = T_{alt}(A_j, A_i)
\]
\end{axiom}

\begin{axiom}[Membrane Preservation]
\[
T_{mem}(m_i, m_j) \Rightarrow m_i = m_j
\]
No chart transition may change membrane category.
\end{axiom}

\begin{axiom}[Citation-Class Preservation]
\[
T_{cls}(c_i, c_j) \Rightarrow c_i = c_j
\]
Citation class must remain invariant under atlas transitions.
\end{axiom}
