\subsection{Altitude Axioms}

\[
A0 < A1 < A2 < A3 < A4 < A5

\subsection{Membrane Axioms}

\begin{axiom}[Neutral-to-Structural]
Neutral $\rightarrow$ Structural is allowed only at A2 or Forest.
\end{axiom}

\begin{axiom}[Structural-to-Formal]
Structural $\rightarrow$ Formal requires citation class $A$ or $B$.
\end{axiom}

\begin{axiom}[Formal-to-Diagnostic]
Formal $\rightarrow$ Diagnostic is allowed only at A4.
\end{axiom}

\begin{axiom}[Diagnostic-to-Provenance]
Diagnostic $\rightarrow$ Provenance is allowed only at A5.
\end{axiom}

\begin{axiom}[Provenance-to-Neutral]
Provenance $\rightarrow$ Neutral requires sovereignty invariants.
\end{axiom}
