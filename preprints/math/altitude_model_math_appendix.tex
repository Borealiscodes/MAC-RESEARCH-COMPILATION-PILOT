\section{Proxy Hilbert Geometry}

\subsection{Definition}
Let $\mathcal{H}' = \mathbb{R}^n$ equipped with the inner product
\[
\langle x, y \rangle' = x^\top y.
\]

\subsection{Norm}
The induced norm is
\[
\|x\|' = \sqrt{x^\top x}.
\]

\subsection{Adjoint}
For any operator $A \in \mathbb{R}^{n \times n}$, the adjoint is defined as
\[
A^{\dagger'} = A^\top.
\]

\subsection{Operator Norm}
\[
\|A\|' = \sup_{\|x\|' = 1} \|Ax\|'.
\]

\section{Density Operators}

\subsection{Definition}
A proxy density operator is any matrix $\rho' \in \mathbb{R}^{n \times n}$ satisfying:
\[
\rho' \ge 0, \qquad \mathrm{Tr}(\rho') = 1.
\]

\subsection{Positivity Condition}
For all $x \in \mathbb{R}^n$,
\[
x^\top \rho' x \ge 0.
\]

\subsection{Spectral Decomposition}
If $\rho'$ is diagonalizable,
\[
\rho' = \sum_{i=1}^n \lambda_i v_i v_i^\top,
\]
where $\lambda_i \ge 0$ and $\sum_i \lambda_i = 1$.
